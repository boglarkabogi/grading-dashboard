\end{align*}
$$

$$
\begin{align*}

&= (x^4+x^3+3x^2+2x+2) + x + 1\\
&= f(x) + x + 1.
\end{align*}
$$

$$
\begin{align*}

&= (x^2+2)(x^2+x+1).
\end{align*}
$$

$$
