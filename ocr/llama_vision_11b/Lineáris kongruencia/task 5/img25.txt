\begin{align*}
13x &\equiv 13 \pmod{409} \\
\Rightarrow\qquad x &\equiv 1 \pmod{409} \\
\end{align*}
\begin{align*}
12x &\equiv 7 \pmod{302} \\
\Rightarrow\qquad x &\equiv 7 \cdot 25 \equiv 175 \equiv 175 \pmod{302} \\
\end{align*}
\begin{align*}
x &\equiv 1 \pmod{409} \\
x &\equiv 175 \pmod{302} \\
\end{align*}
