$f=x^{4}+2 x^{3}+x^{2}+4 \in \mathbb{Z}_{5}[x]$

$g=x^{5}+x^{4}+2 x^{3}+2 x^{2}+2 x+1 \in \mathbb{Z}_{5}[x]$

$r_{1}=g-x f=-x^{4}+x^{3}+2 x^{2}-2 x+1$

$\equiv 4 x^{4}+x^{3}+2 x^{2}+3 x+1$

$4 \cdot r_{1}=16 x^{4}+4 x^{3}+8 x^{2}+12 x+4 \equiv 1 x^{4}+4 x^{3}+3 x^{2}+2 x+4$

$r_{2}=f-4 r_{1}=-2 x^{3}-2 x^{2}-2 x \equiv 3 x^{3}+3 x^{2}+3 x$

$4 / 3 \bmod 5 \equiv 3$

$r_{3}=r_{1}-3 x r_{2}=-3 x^{3}-2 x^{2}+3 x+1$

$\equiv 2 x^{3}+3 x^{2}+3 x+1$

$3 / 2 \bmod 5=4$

$r_{4}=r_{2}-4 r_{3}=x^{2}+x-4$

$\equiv x^{2}+x+1$

$r_{3}-2 x r_{4}=x^{2}+x+1=r_{4}$

$r_{4}=r_{5}$
