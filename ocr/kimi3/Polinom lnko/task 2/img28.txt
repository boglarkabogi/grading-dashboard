**6. Beadandó**

\[
f=x^{4}+x^{3}+4x+4\in \mathbb{Z}_{5}[x]
\qquad
g=x^{5}+x^{4}+4x^{3}+2\in \mathbb{Z}_{5}[x]
\]

**(1)**  
\[
g=q_1\cdot f+r_1
\]
\[
g_1=g-(x\cdot f)=g-(x^{5}+x^{4}+4x^{2}+4x)=4x^{3}-4x^{2}-4x+2
\]
\[
\deg g_1<\deg f \Rightarrow \text{állj} \Rightarrow
q_1=x
\qquad
r_1=4x^{3}-4x^{2}-4x+2
\]

**(2)**  
\[
f=q_2\cdot r_1+r_2
\]
\[
f_1=f-(4x\cdot r_1)=f-(x^{4}-x^{3}-x^{2}+3x)=2x^{3}+x^{2}+x+4
\]
\[
\deg f_1=\deg r_1
\]
\[
f_2=f_1-(3\cdot r_1)=f_1-(2x^{3}-2x^{2}-2x+1)=3x^{2}+3x+3
\]
\[
\deg f_2<\deg r_1 \Rightarrow \text{állj} \Rightarrow
q_2=4x+3
\qquad
r_2=3x^{2}+3x+3
\]

**(3)**  
\[
r_1=q_3\cdot r_2+r_3
\]
\[
(r_1)_1=r_1-(3x\cdot r_2)=r_1-(4x^{3}+4x^{2}+4x)=3x^{2}+3x+2
\]
\[
(r_1)_2=(r_1)_1-(1\cdot r_2)=(r_1)_1-(3x^{2}+3x+3)=-1
\]
\[
\deg (r_1)_2<\deg r_2 \Rightarrow \text{állj} \Rightarrow
q_3=3x+1
\qquad
r_3=-1
\]

**(4)**  
\[
r_2=q_4\cdot r_3+r_4
\]
\[
(r_2)_1=r_2-\left(-3x^{2}\cdot r_3\right)=r_2-\left(3x^{2}\right)=3x+3
\]
\[
(r_2)_2=(r_2)_1-\left(-3x\cdot r_3\right)=(r_2)_1-(3x)=3
\]
\[
(r_2)_3=(r_2)_2-\left(-3\cdot r_3\right)=(r_2)_2-3=0
\]

\[
\text{Tehát } \operatorname{lnko}(f,g)=-1
\]
