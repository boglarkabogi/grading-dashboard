$f = x^4 + 3x^3 + 2x^2 + x + 4 \in \mathbb{Z}_5[x]$  
$g = x^5 + x^4 + x^2 + x + 2 \in \mathbb{Z}_5[x]$

$g = \boxed{?} \cdot f + r_1$

$x^5+x^4+x^2+x+2 : x^4+3x^3+2x^2+x+4 = x$  
$\underline{x^5+3x^4+2x^3+x^2+4x \quad \ominus}$  
$3x^4+3x^3+2x+2 \quad (r_1)$

$g = x \cdot f + 3x^4+3x^3+2x+2$  
$f = \boxed{?} \cdot r_1 + r_2$

$x^4+3x^3+2x^2+x+4 : 3x^4+3x^3+2x+2 = 2$  
$\underline{x^4+x^3+4x+4 \quad \ominus}$  
$2x^3+2x^2+2x \quad (r_2)$

$f = 2x^3+2x^2+2x \cdot r_1 + r_2$  
$r_1 = \boxed{?} r_2 + r_3$

$3x^4+3x^3+2x+2 : 2x^3+2x^2+2x = 4x$  
$\underline{3x^4+3x^3+3x^2 \quad \ominus}$  
$2x^2+2x+2 \quad (r_3)$

$r_1 = 4x \cdot r_2 + r_3$  
$r_2 = \boxed{?} r_3 + r_4$

$2x^3+2x^2+2x : 2x^2+2x+2 = x$  
$\underline{2x^3+2x^2+2x \quad \ominus}$  
$0 \quad (r_4)$

$r_2 = x \cdot r_3 + r_4 \rightarrow 0$

$\mathrm{lnko}(f,g) = r_3 = 2x^2+2x+2$
