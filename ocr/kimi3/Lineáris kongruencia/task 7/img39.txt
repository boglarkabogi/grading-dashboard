\[
13x \equiv 19 \mod 414
\]

\[
13x+414y=19
\]

\[
\begin{array}{c|c|c|c|c}
-1 & 414 & - & 0 & 1\\
0 & 13 & - & 1 & 0\\
1 & 11 & 31 & -31 & 1\\
2 & 2 & 1 & 32 & -1\\
3 & 1 & 5 & -191 & 6
\end{array}
\qquad
\begin{aligned}
0\cdot 13+1\cdot 414&=414\\
1\cdot 13+0\cdot 414&=13\\
-31\cdot 13+1\cdot 414&=11\\
32\cdot 13+(-1)\cdot 414&=2\\
-191\cdot 13+6\cdot 414&=1
\end{aligned}
\]

\[
x'=-191 \qquad y'=6
\]

\[
x_0=-191\cdot \frac{19}{1}=3629
\]

\[
y_0=6\cdot \frac{19}{1}=114
\]

\[
x_t=3629+\frac{414}{1}\cdot t
\]

\[
y_t=114+\frac{19}{1}\cdot t
\]

\[
20x\equiv 6 \mod 412 \Rightarrow 20x+412y=6
\]

\[
\begin{array}{c|c|c|c|c}
-1 & 412 & - & 0 & 1\\
1 & 20 & - & 1 & 0\\
2 & 12 & 20 & -20 & 1\\
3 & 8 & 1 & 21 & -1\\
4 & 4 & 2 & -62 & 3
\end{array}
\qquad
\begin{aligned}
0\cdot 20+1\cdot 412&=412\\
1\cdot 20+0\cdot 412&=20\\
-20\cdot 20+1\cdot 412&=12\\
21\cdot 20+(-1)\cdot 412&=8\\
-62\cdot 20+3\cdot 412&=4
\end{aligned}
\]

\[
x'=-62 \qquad y'=3
\]

\[
x_0=-62\cdot \frac{6}{4}=-93
\]

\[
y_0=3\cdot \frac{6}{4}=\frac{9}{2}\notin \mathbb{N} \Rightarrow \text{Nincs mo.}
\]
