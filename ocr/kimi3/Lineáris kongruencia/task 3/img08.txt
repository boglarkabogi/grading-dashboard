\[
11x \equiv 2 \mod 238
\]

| \(238\) | \(1\) | \(0\) |
|---|---|---|
| \(11\) | \(0\) | \(1\) |

\[
238-11\cdot 21=7 \qquad 1-21\cdot 0=1 \qquad 0-21\cdot 1=-21
\]

\[
11-7\cdot 1=4 \qquad 0-1\cdot 1=-1 \qquad 1-1\cdot(-21)=22
\]

\[
7-4\cdot 1=3 \qquad 1-1\cdot 1 \qquad -21-1\cdot 22=-43
\]

\[
4-3\cdot 1=\boxed{1} \qquad 1-1\cdot 3=-3 \qquad 22-1\cdot(-43)=65
\]

\[
3-1\cdot 3=0
\]

\[
1=65\cdot 11-3\cdot 238
\]

\[
65\cdot 11\equiv 1 \mod 238
\]

\[
130\cdot 11\equiv 2 \mod 238
\]

\[

|---|---|---|

\]

\[
16-10\cdot 1=6
\]

\[

\]

\[
