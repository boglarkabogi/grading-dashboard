f = x^4 + 1 \in \mathbb{Z}_5[x]

g = x^3 + 2x \in \mathbb{Z}_5[x]

x^2 \equiv 4x + 1 \pmod{x^3 + x + 4}

f \equiv 4x^2 + x + 1 \pmod{x^3 + x + 4}

g \equiv x + 1 \pmod{x^3 + x + 4}

a) f \cdot g \equiv 4x^3 + 4x^2 + x^2 + x + x + 1 = 4x^3 + 2x + 1 \equiv 3x \pmod{x^3 + x + 4}

b) f \cdot g^2 \cdot f^2 \equiv

f^2 \equiv 16x^4 + 8x^3 + 8x^2 + 2x^2 + 2x + 1

\equiv 4x^2 + x + 12x^2 + 3 + 3x^2 + 2x^2 + 2x + 1

\equiv 4x^2 + 4 \pmod{x^3 + x + 4}

f \cdot g^2 + f^2 \cdot g \equiv 4x^4 + 4x^3 + 2x^2 + 2x + 1

(x+1)(x^5 + x^4 + x + 1) + 4x^3 + 4x^2 + 4x + 4

x^2 + 2x^5 + x^4 + x^2 + 2x + 1

= x^2 \cdot x^3 + 2x^2 \cdot x^3 + x \cdot x^3 + x^2 + 2x + 1

\equiv 4x^4 + x^3 + 8x^3 + 2x^2 + 4x^2 + x + x^2 + 2x + 1

\equiv 16x^4 + 20x + 4 + 2x^2 + 3x + 1

\equiv 3x^2 + 3x \pmod{x^3 + x + 4}

c) f^4 \equiv 16x^4 + 32x^2 + 16 \equiv 6x^2 + x + 1 \pmod{x^3 + x + 4}

f^8 \equiv x^4 + x^2 + 1 + 2x^3 + 2x^2 + 2x

= x^4 + 8x^4 + 2x^3 + 3x^2 + 2x + 1 \equiv 4x^2 + x + 8x + 2 + 3x^2 + 2x + 1

\equiv 7x^2 + 8x + 3 \pmod{x^3 + x + 4}
