\[

\]

\[
v \cdot (x^2 - 1) \equiv -x^2 + 2x + 3 \quad (x^2 - 2x + 3)
\]

\[
+x^2 - 2x + 3 \equiv 0 \quad (x^2 - 2x + 3)
\]

\[
x^2 \equiv 2x - 3 \quad (x^2 - 2x + 3)
\]

\[
v \cdot (2x - 4) \equiv 6 \quad (x^2 - 2x + 3)
\]

\[
g = x^2 - 1 \equiv 2x - 4 \quad (x^2 - 2x + 3)
\]

\[
h = -x^2 + 2x + 3 \equiv -(2x - 4) + 2x + 3 \quad (f)
\]

\[
h \equiv 6
\]
