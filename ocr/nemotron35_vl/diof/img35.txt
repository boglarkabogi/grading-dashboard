u = ax + b \quad v = cx + d

(ax + b) \cdot (x^2 + 3x + 3) + (cx + d) \cdot (x^2 - 1) = 2x^2 + x - 3

ax^3 + 3ax^2 + 3ax + bx^2 + 3bx + 3b + cx^3 - cx + dx^2 - d = 2x^2 + x - 3

(a + c)x^3 + (3a + b + d)x^2 + (3a + 3b - c)x + 3b - d = 2x^2 + x - 3

\begin{cases}

3a + b + d = 2 \\
3a + 3b - c = 1 \\
3b - d = -3 \rightarrow d = 3b + 3
\end{cases}
