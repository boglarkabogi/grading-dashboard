u \cdot (x^2 + x + 1) + v \cdot (x^2 - 1) = x - 1 \quad u, v \in \mathbb{Q}[x]

f = x^2 + x + 1 \quad g = x^2 - 1

\cancel{f} = x^2 + x + 1 = x^2 - 1 + (x + 2) \rightarrow f(x) - g(x) = x + 2

x^2 - 1 = (x - 2)(x + 2) + 3 \rightarrow 3 = g(x) - (x + 2)(x - 2)

\text{holds the truth ...} \quad x + 2 = f(x) - g(x)

3 = g(x) - (x - 2)(f(x) - g(x))

3 = -(x - 2) \cdot f(x) + (x - 1) \cdot g(x) \quad /:3

1 = \frac{2 - x}{3} f(x) + \frac{x - 1}{3} g(x) \quad / \cdot (x - 1)

x - 1 = \frac{(2 - x)(x - 1)}{3} f(x) + \frac{(x - 1)^2}{3} g(x)

\text{then } u = \frac{(2 - x)(x - 1)}{3} \quad v = \frac{(x - 1)^2}{3}
